    41→function useFavorites() {
    42→  const [favs, setFavs] = useState<string[]>(() => {
    43→    if (typeof window === 'undefined') return [];
    44→    try { return JSON.parse(localStorage.getItem(FAVS_KEY) || '[]'); } catch { return []; }
    45→  });
    46→  const toggle = useCallback((id: string) => {
    47→    setFavs(prev => {
    48→      const next = prev.includes(id) ? prev.filter(f => f !== id) : [...prev, id];
    49→      localStorage.setItem(FAVS_KEY, JSON.stringify(next));
    50→      return next;
    51→    });
    52→  }, []);
    53→  const isFav = useCallback((id: string) => favs.includes(id), [favs]);
    54→  return { favs, toggle, isFav };
    55→}

    80→function getHarmonyScore(colors: string[]): { score: number; label: string; description: string } {
    81→  if (colors.length < 2) return { score: 50, label: 'Neutro', description: 'Datos insuficientes' };
    82→  
    83→  const hexToHSL = (hex: string) => {
    84→    let r = parseInt(hex.slice(1, 3), 16) / 255;
    85→    let g = parseInt(hex.slice(3, 5), 16) / 255;
    86→    let b = parseInt(hex.slice(5, 7), 16) / 255;
    87→    const max = Math.max(r, g, b), min = Math.min(r, g, b);
    88→    let h = 0, s = 0, l = (max + min) / 2;
    89→    if (max !== min) {
    90→      const d = max - min;
    91→      s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
    92→      if (max === r) h = ((g - b) / d + (g < b ? 6 : 0)) / 6;
    93→      else if (max === g) h = ((b - r) / d + 2) / 6;
    94→      else h = ((r - g) / d + 4) / 6;
    95→    }
    96→    return { h: h * 360, s: s * 100, l: l * 100 };
    97→  };
    98→
    99→  const hueDiff = (a: number, b: number) => Math.abs(((a - b + 540) % 360) - 180);
   100→
   101→  const hslColors = colors.map(hexToHSL);
   102→  const neutrals = hslColors.filter(c => c.s < 15);
   103→  const chromatics = hslColors.filter(c => c.s >= 15);
   104→
   105→  let score = 60; // base
   106→
   107→  // Bonus for neutrals (they go with everything)
   108→  score += Math.min(neutrals.length * 5, 10);
   109→
   110→  // Chromatic harmony
   111→  if (chromatics.length >= 2) {
   112→    const maxHueDiff = Math.max(...chromatics.flatMap((c, i) => chromatics.slice(i + 1).map(d => hueDiff(c.h, d.h))));
   113→    const avgSat = chromatics.reduce((sum, c) => sum + c.s, 0) / chromatics.length;
   114→    const avgLight = chromatics.reduce((sum, c) => sum + c.l, 0) / chromatics.length;
   115→
   116→    if (maxHueDiff < 30 || maxHueDiff > 330) score += 15; // Analogous
   117→    else if (Math.abs(maxHueDiff - 180) < 30) score += 12; // Complementary
   118→    else if (maxHueDiff > 90 && maxHueDiff < 270) score += 8; // Triadic area
   119→    else score += 5;
   120→
   121→    if (avgSat < 40) score += 5; // Muted palette bonus
   122→    if (avgLight > 20 && avgLight < 80) score += 5; // Good lightness range
   123→  }
   124→
   125→  // Monochrome bonus (very intentional)
   126→  const uniqueHues = new Set(hslColors.filter(c => c.s >= 15).map(c => Math.round(c.h / 15) * 15));
   127→  if (uniqueHues.size <= 2 && hslColors.length >= 3) score += 8;
   128→
   129→  score = Math.min(Math.max(score, 30), 98);
   130→
   131→  if (score >= 85) return { score, label: 'Armonia Excelente', description: 'Paleta cohesiva y sofisticada' };
   132→  if (score >= 70) return { score, label: 'Buena Armonia', description: 'Combinacion equilibrada y atractiva' };
   133→  if (score >= 55) return { score, label: 'Armonia Moderada', description: 'Funcional con potencial de mejora' };
   134→  return { score, label: 'Contraste Alto', description: 'Combinacion atrevida y audaz' };
   135→}

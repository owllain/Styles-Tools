    85→function getHarmonyScore(colors: string[]): { score: number; label: string; description: string } {
    86→  if (colors.length < 2) return { score: 50, label: 'Neutro', description: 'Datos insuficientes' };
    87→  
    88→  const hexToHSL = (hex: string) => {
    89→    let r = parseInt(hex.slice(1, 3), 16) / 255;
    90→    let g = parseInt(hex.slice(3, 5), 16) / 255;
    91→    let b = parseInt(hex.slice(5, 7), 16) / 255;
    92→    const max = Math.max(r, g, b), min = Math.min(r, g, b);
    93→    let h = 0, s = 0, l = (max + min) / 2;
    94→    if (max !== min) {
    95→      const d = max - min;
    96→      s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
    97→      if (max === r) h = ((g - b) / d + (g < b ? 6 : 0)) / 6;
    98→      else if (max === g) h = ((b - r) / d + 2) / 6;
    99→      else h = ((r - g) / d + 4) / 6;
   100→    }
   101→    return { h: h * 360, s: s * 100, l: l * 100 };
   102→  };
   103→
   104→  const hueDiff = (a: number, b: number) => Math.abs(((a - b + 540) % 360) - 180);
   105→
   106→  const hslColors = colors.map(hexToHSL);
   107→  const neutrals = hslColors.filter(c => c.s < 15);
   108→  const chromatics = hslColors.filter(c => c.s >= 15);
   109→
   110→  let score = 60; // base
   111→
   112→  // Bonus for neutrals (they go with everything)
   113→  score += Math.min(neutrals.length * 5, 10);
   114→
   115→  // Chromatic harmony
   116→  if (chromatics.length >= 2) {
   117→    const maxHueDiff = Math.max(...chromatics.flatMap((c, i) => chromatics.slice(i + 1).map(d => hueDiff(c.h, d.h))));
   118→    const avgSat = chromatics.reduce((sum, c) => sum + c.s, 0) / chromatics.length;
   119→    const avgLight = chromatics.reduce((sum, c) => sum + c.l, 0) / chromatics.length;
   120→
   121→    if (maxHueDiff < 30 || maxHueDiff > 330) score += 15; // Analogous
   122→    else if (Math.abs(maxHueDiff - 180) < 30) score += 12; // Complementary
   123→    else if (maxHueDiff > 90 && maxHueDiff < 270) score += 8; // Triadic area
   124→    else score += 5;
   125→
   126→    if (avgSat < 40) score += 5; // Muted palette bonus
   127→    if (avgLight > 20 && avgLight < 80) score += 5; // Good lightness range
   128→  }
   129→
   130→  // Monochrome bonus (very intentional)
   131→  const uniqueHues = new Set(hslColors.filter(c => c.s >= 15).map(c => Math.round(c.h / 15) * 15));
   132→  if (uniqueHues.size <= 2 && hslColors.length >= 3) score += 8;
   133→
   134→  score = Math.min(Math.max(score, 30), 98);
   135→
   136→  if (score >= 85) return { score, label: 'Armonia Excelente', description: 'Paleta cohesiva y sofisticada' };
   137→  if (score >= 70) return { score, label: 'Buena Armonia', description: 'Combinacion equilibrada y atractiva' };
   138→  if (score >= 55) return { score, label: 'Armonia Moderada', description: 'Funcional con potencial de mejora' };
   139→  return { score, label: 'Contraste Alto', description: 'Combinacion atrevida y audaz' };
   140→}

  2206→function StyleDNARadar({ favs, worn, ratings, allOutfits }: { favs: string[]; worn?: Record<string, WornEntry>; ratings?: Record<string, number>; allOutfits: OutfitBasic[] }) {
  2207→  const estilos = ['noir', 'old_money', 'rockero', 'corporate'] as const;
  2208→  const estiloLabels: Record<string, string> = { noir: 'Noir', old_money: 'Old Money', rockero: 'Rockero', corporate: 'Corporate' };
  2209→  const estiloColors: Record<string, string> = { noir: '#9ca3af', old_money: '#f59e0b', rockero: '#ef4444', corporate: '#38bdf8' };
  2210→
  2211→  const scores = useMemo(() => {
  2212→    const counts: Record<string, number> = { noir: 0, old_money: 0, rockero: 0, corporate: 0 };
  2213→    const weightedCounts: Record<string, number> = { noir: 0, old_money: 0, rockero: 0, corporate: 0 };
  2214→
  2215→    allOutfits.forEach(o => {
  2216→      o.estilo.forEach(e => { if (counts[e] !== undefined) counts[e]++; });
  2217→    });
  2218→
  2219→    // Weight: favorites = 5pts, worn = 3pts, rated = 4pts
  2220→    favs.forEach(id => {
  2221→      const o = allOutfits.find(x => x.id === id);
  2222→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 5; });
  2223→    });
  2224→    Object.entries(worn || {}).forEach(([id, entry]) => {
  2225→      const o = allOutfits.find(x => x.id === id);
  2226→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 3 * entry.count; });
  2227→    });
  2228→    Object.entries(ratings || {}).forEach(([id, stars]) => {
  2229→      const o = allOutfits.find(x => x.id === id);
  2230→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 4 * (stars / 5); });
  2231→    });
  2232→
  2233→    // If no interactions, use base distribution
  2234→    const hasInteractions = favs.length > 0 || Object.keys(worn).length > 0 || Object.keys(ratings).length > 0;
  2235→    if (!hasInteractions) {
  2236→      const total = Object.values(counts).reduce((a, b) => a + b, 0);
  2237→      return estilos.map(e => total > 0 ? (counts[e] / total) * 100 : 25);
  2238→    }

  2240→    const total = Object.values(weightedCounts).reduce((a, b) => a + b, 0);
  2241→    if (total === 0) return estilos.map(() => 25);

  2259→
  2260→  const dominantIndex = scores.indexOf(Math.max(...scores));
  2261→  const dominant = estilos[dominantIndex];

  2268→          <svg viewBox="0 0 100 100" className="w-40 h-40" overflow="visible">
  2269→            {/* Grid lines */}

  2271→              const pts = estilos.map((_, i) => getPoint(i, level));
  2272→              const path = pts.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
  2273→              return <path key={level} d={path} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth="0.3" />;

  2275→            {/* Axis lines */}
  2276→            {estilos.map((_, i) => {
  2277→              const p = getPoint(i, 100);
  2278→              return <line key={i} x1={center} y1={center} x2={p.x} y2={p.y} stroke="rgba(255,255,255,0.03)" strokeWidth="0.3" />;

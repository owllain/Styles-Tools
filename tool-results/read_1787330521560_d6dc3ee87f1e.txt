  2269→function StyleDNARadar({ favs, worn, ratings, allOutfits }: { favs: string[]; worn?: Record<string, WornEntry>; ratings?: Record<string, number>; allOutfits: OutfitBasic[] }) {
  2270→  const estilos = ['noir', 'old_money', 'rockero', 'corporate'] as const;
  2271→  const estiloLabels: Record<string, string> = { noir: 'Noir', old_money: 'Old Money', rockero: 'Rockero', corporate: 'Corporate' };
  2272→  const estiloColors: Record<string, string> = { noir: '#9ca3af', old_money: '#f59e0b', rockero: '#ef4444', corporate: '#38bdf8' };
  2273→
  2274→  const scores = useMemo(() => {
  2275→    const counts: Record<string, number> = { noir: 0, old_money: 0, rockero: 0, corporate: 0 };
  2276→    const weightedCounts: Record<string, number> = { noir: 0, old_money: 0, rockero: 0, corporate: 0 };
  2277→
  2278→    allOutfits.forEach(o => {
  2279→      o.estilo.forEach(e => { if (counts[e] !== undefined) counts[e]++; });
  2280→    });
  2281→
  2282→    // Weight: favorites = 5pts, worn = 3pts, rated = 4pts
  2283→    favs.forEach(id => {
  2284→      const o = allOutfits.find(x => x.id === id);
  2285→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 5; });
  2286→    });
  2287→    Object.entries(worn || {}).forEach(([id, entry]) => {
  2288→      const o = allOutfits.find(x => x.id === id);
  2289→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 3 * entry.count; });
  2290→    });
  2291→    Object.entries(ratings || {}).forEach(([id, stars]) => {
  2292→      const o = allOutfits.find(x => x.id === id);
  2293→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 4 * (stars / 5); });
  2294→    });
  2295→
  2296→    // If no interactions, use base distribution
  2297→    const hasInteractions = favs.length > 0 || Object.keys(worn).length > 0 || Object.keys(ratings).length > 0;
  2298→    if (!hasInteractions) {
  2299→      const total = Object.values(counts).reduce((a, b) => a + b, 0);
  2300→      return estilos.map(e => total > 0 ? (counts[e] / total) * 100 : 25);
  2301→    }
  2302→
  2303→    const total = Object.values(weightedCounts).reduce((a, b) => a + b, 0);
  2304→    if (total === 0) return estilos.map(() => 25);
  2305→    return estilos.map(e => (weightedCounts[e] / total) * 100);
  2306→  }, [favs, worn, ratings, allOutfits]);
  2307→
  2308→  const maxScore = Math.max(...scores, 1);
  2309→  const center = 50;
  2310→  const radius = 38;

  2313→

  2315→    const angle = angleStep * i - Math.PI / 2;
  2316→    const r = (value / 100) * radius;

  2318→  };

  2321→  const dataPath = dataPoints.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
  2322→
  2323→  const dominantIndex = scores.indexOf(Math.max(...scores));
  2324→  const dominant = estilos[dominantIndex];
  2325→
  2326→  return (
  2327→    <Card className="bg-white/[0.015] border-white/[0.04] rounded-2xl p-5 backdrop-blur-sm">
  2328→      <h3 className="text-xs font-semibold text-white/50 uppercase tracking-wider mb-4 flex items-center gap-2"><Gem className="h-3.5 w-3.5 text-violet-400" />Style DNA</h3>
  2329→      <div className="flex flex-col sm:flex-row items-center gap-6">
  2330→        <div className="style-dna-chart relative flex-shrink-0 py-3">
  2331→          <svg viewBox="0 0 100 100" className="w-40 h-40" overflow="visible">
  2332→            {/* Grid lines */}
  2333→            {[25, 50, 75, 100].map(level => {
  2334→              const pts = estilos.map((_, i) => getPoint(i, level));
  2335→              const path = pts.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
  2336→              return <path key={level} d={path} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth="0.3" />;
  2337→            })}
  2338→            {/* Axis lines */}
  2339→            {estilos.map((_, i) => {
  2340→              const p = getPoint(i, 100);
  2341→              return <line key={i} x1={center} y1={center} x2={p.x} y2={p.y} stroke="rgba(255,255,255,0.03)" strokeWidth="0.3" />;

  2346→              fill="rgba(212,168,67,0.15)"
  2347→              stroke="rgba(212,168,67,0.6)"

  2367→              const lr = radius + 10;
  2368→              const lx = center + lr * Math.cos(angle);

  2372→                  {estiloLabels[e]}

  2155→function StyleDNARadar({ favs, worn, ratings, allOutfits }: { favs: string[]; worn?: Record<string, WornEntry>; ratings?: Record<string, number>; allOutfits: OutfitBasic[] }) {
  2156→  const estilos = ['noir', 'old_money', 'rockero', 'corporate'] as const;
  2157→  const estiloLabels: Record<string, string> = { noir: 'Noir', old_money: 'Old Money', rockero: 'Rockero', corporate: 'Corporate' };
  2158→  const estiloColors: Record<string, string> = { noir: '#9ca3af', old_money: '#f59e0b', rockero: '#ef4444', corporate: '#38bdf8' };
  2159→
  2160→  const scores = useMemo(() => {
  2161→    const counts: Record<string, number> = { noir: 0, old_money: 0, rockero: 0, corporate: 0 };
  2162→    const weightedCounts: Record<string, number> = { noir: 0, old_money: 0, rockero: 0, corporate: 0 };
  2163→
  2164→    allOutfits.forEach(o => {
  2165→      o.estilo.forEach(e => { if (counts[e] !== undefined) counts[e]++; });
  2166→    });
  2167→
  2168→    // Weight: favorites = 5pts, worn = 3pts, rated = 4pts
  2169→    favs.forEach(id => {
  2170→      const o = allOutfits.find(x => x.id === id);
  2171→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 5; });
  2172→    });
  2173→    Object.entries(worn || {}).forEach(([id, entry]) => {
  2174→      const o = allOutfits.find(x => x.id === id);
  2175→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 3 * entry.count; });
  2176→    });
  2177→    Object.entries(ratings || {}).forEach(([id, stars]) => {
  2178→      const o = allOutfits.find(x => x.id === id);
  2179→      if (o) o.estilo.forEach(e => { if (weightedCounts[e] !== undefined) weightedCounts[e] += 4 * (stars / 5); });
  2180→    });
  2181→
  2182→    // If no interactions, use base distribution
  2183→    const hasInteractions = favs.length > 0 || Object.keys(worn).length > 0 || Object.keys(ratings).length > 0;
  2184→    if (!hasInteractions) {
  2185→      const total = Object.values(counts).reduce((a, b) => a + b, 0);
  2186→      return estilos.map(e => total > 0 ? (counts[e] / total) * 100 : 25);
  2187→    }
  2188→
  2189→    const total = Object.values(weightedCounts).reduce((a, b) => a + b, 0);
  2190→    if (total === 0) return estilos.map(() => 25);
  2191→    return estilos.map(e => (weightedCounts[e] / total) * 100);
  2192→  }, [favs, worn, ratings, allOutfits]);
  2193→
  2194→  const maxScore = Math.max(...scores, 1);
  2195→  const center = 50;
  2196→  const radius = 38;
  2197→  const n = estilos.length;
  2198→  const angleStep = (2 * Math.PI) / n;
  2199→
  2200→  const getPoint = (i: number, value: number) => {
  2201→    const angle = angleStep * i - Math.PI / 2;
  2202→    const r = (value / 100) * radius;
  2203→    return { x: center + r * Math.cos(angle), y: center + r * Math.sin(angle) };
  2204→  };
  2205→
  2206→  const dataPoints = scores.map((s, i) => getPoint(i, s));
  2207→  const dataPath = dataPoints.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
  2208→
  2209→  const dominantIndex = scores.indexOf(Math.max(...scores));
  2210→  const dominant = estilos[dominantIndex];
  2211→
  2212→  return (
  2213→    <Card className="bg-white/[0.015] border-white/[0.04] rounded-2xl p-5 backdrop-blur-sm">
  2214→      <h3 className="text-xs font-semibold text-white/50 uppercase tracking-wider mb-4 flex items-center gap-2"><Gem className="h-3.5 w-3.5 text-violet-400" />Style DNA</h3>
  2215→      <div className="flex flex-col sm:flex-row items-center gap-6">
  2216→        <div className="style-dna-chart relative flex-shrink-0 py-3">
  2217→          <svg viewBox="0 0 100 100" className="w-40 h-40" overflow="visible">
  2218→            {/* Grid lines */}
  2219→            {[25, 50, 75, 100].map(level => {
  2220→              const pts = estilos.map((_, i) => getPoint(i, level));
  2221→              const path = pts.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ') + ' Z';
  2222→              return <path key={level} d={path} fill="none" stroke="rgba(255,255,255,0.04)" strokeWidth="0.3" />;
  2223→            })}
  2224→            {/* Axis lines */}
  2225→            {estilos.map((_, i) => {
  2226→              const p = getPoint(i, 100);
  2227→              return <line key={i} x1={center} y1={center} x2={p.x} y2={p.y} stroke="rgba(255,255,255,0.03)" strokeWidth="0.3" />;
  2228→            })}
  2229→            {/* Data polygon */}
  2230→            <motion.path
  2231→              d={dataPath}
  2232→              fill="rgba(212,168,67,0.08)"
  2233→              stroke="rgba(212,168,67,0.5)"
  2234→              strokeWidth="0.8"
  2235→              initial={{ opacity: 0, scale: 0.5 }}
  2236→              animate={{ opacity: 1, scale: 1 }}
  2237→              transition={{ duration: 0.8, type: 'spring' }}
  2238→              style={{ transformOrigin: `${center}px ${center}px` }}
  2239→            />
  2240→            {/* Data points */}
  2241→            {dataPoints.map((p, i) => (
  2242→              <motion.circle
  2243→                key={i} cx={p.x} cy={p.y} r="1.8"
  2244→                fill={estiloColors[estilos[i]]}
  2245→                stroke="#08080a" strokeWidth="0.8"

  2247→              />
  2248→            ))}
  2249→            {/* Labels */}
  2250→            {estilos.map((e, i) => {
  2251→              const angle = angleStep * i - Math.PI / 2;
  2252→              const lr = radius + 10;
  2253→              const lx = center + lr * Math.cos(angle);
  2254→              const ly = center + lr * Math.sin(angle);
  2255→              return (
  2256→                <text key={e} x={lx} y={ly} textAnchor="middle" dominantBaseline="middle" className="text-[5px] font-semibold" fill={scores[i] === maxScore ? estiloColors[e] : 'rgba(255,255,255,0.25)'}>
  2257→                  {estiloLabels[e]}
  2258→                </text>
  2259→              );
  2260→            })}
  2261→          </svg>
  2262→        </div>
  2263→        <div className="flex-1 space-y-3">
  2264→          <div className="flex items-center gap-2 px-3 py-2 rounded-xl border border-white/[0.04] bg-white/[0.015]">
  2265→            <span className="text-lg">{dominant === 'noir' ? '🖤' : dominant === 'old_money' ? '👑' : dominant === 'rockero' ? '🎸' : '💼'}</span>

  2267→              <p className="text-[11px] font-semibold text-white/70">Estilo Dominante</p>

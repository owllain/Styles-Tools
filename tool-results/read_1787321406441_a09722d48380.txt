    49→function useFavorites() {
    50→  const [favs, setFavs] = useState<string[]>(() => {
    51→    if (typeof window === 'undefined') return [];
    52→    try { return JSON.parse(localStorage.getItem(FAVS_KEY) || '[]'); } catch { return []; }
    53→  });
    54→  const toggle = useCallback((id: string) => {
    55→    setFavs(prev => {
    56→      const next = prev.includes(id) ? prev.filter(f => f !== id) : [...prev, id];
    57→      localStorage.setItem(FAVS_KEY, JSON.stringify(next));
    58→      return next;
    59→    });
    60→  }, []);
    61→  const isFav = useCallback((id: string) => favs.includes(id), [favs]);
    62→  return { favs, toggle, isFav };
    63→}

    89→      const updated = { ...entry, count: entry.count + 1, lastWorn: today, dates: [...new Set([...entry.dates, today])] };

    93→    });

   121→function getHarmonyScore(colors: string[]): { score: number; label: string; description: string } {
   122→  if (colors.length < 2) return { score: 50, label: 'Neutro', description: 'Datos insuficientes' };
   123→  
   124→  const hexToHSL = (hex: string) => {
   125→    let r = parseInt(hex.slice(1, 3), 16) / 255;
   126→    let g = parseInt(hex.slice(3, 5), 16) / 255;
   127→    let b = parseInt(hex.slice(5, 7), 16) / 255;
   128→    const max = Math.max(r, g, b), min = Math.min(r, g, b);
   129→    let h = 0, s = 0, l = (max + min) / 2;
   130→    if (max !== min) {
   131→      const d = max - min;
   132→      s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
   133→      if (max === r) h = ((g - b) / d + (g < b ? 6 : 0)) / 6;
   134→      else if (max === g) h = ((b - r) / d + 2) / 6;
   135→      else h = ((r - g) / d + 4) / 6;
   136→    }
   137→    return { h: h * 360, s: s * 100, l: l * 100 };
   138→  };
   139→
   140→  const hueDiff = (a: number, b: number) => Math.abs(((a - b + 540) % 360) - 180);
   141→
   142→  const hslColors = colors.map(hexToHSL);
   143→  const neutrals = hslColors.filter(c => c.s < 15);
   144→  const chromatics = hslColors.filter(c => c.s >= 15);
   145→
   146→  let score = 60; // base
   147→
   148→  // Bonus for neutrals (they go with everything)
   149→  score += Math.min(neutrals.length * 5, 10);
   150→
   151→  // Chromatic harmony
   152→  if (chromatics.length >= 2) {
   153→    const maxHueDiff = Math.max(...chromatics.flatMap((c, i) => chromatics.slice(i + 1).map(d => hueDiff(c.h, d.h))));
   154→    const avgSat = chromatics.reduce((sum, c) => sum + c.s, 0) / chromatics.length;
   155→    const avgLight = chromatics.reduce((sum, c) => sum + c.l, 0) / chromatics.length;
   156→
   157→    if (maxHueDiff < 30 || maxHueDiff > 330) score += 15; // Analogous
   158→    else if (Math.abs(maxHueDiff - 180) < 30) score += 12; // Complementary
   159→    else if (maxHueDiff > 90 && maxHueDiff < 270) score += 8; // Triadic area
   160→    else score += 5;
   161→
   162→    if (avgSat < 40) score += 5; // Muted palette bonus
   163→    if (avgLight > 20 && avgLight < 80) score += 5; // Good lightness range
   164→  }
   165→
   166→  // Monochrome bonus (very intentional)
   167→  const uniqueHues = new Set(hslColors.filter(c => c.s >= 15).map(c => Math.round(c.h / 15) * 15));
   168→  if (uniqueHues.size <= 2 && hslColors.length >= 3) score += 8;
   169→
   170→  score = Math.min(Math.max(score, 30), 98);
   171→
   172→  if (score >= 85) return { score, label: 'Armonia Excelente', description: 'Paleta cohesiva y sofisticada' };
   173→  if (score >= 70) return { score, label: 'Buena Armonia', description: 'Combinacion equilibrada y atractiva' };
   174→  if (score >= 55) return { score, label: 'Armonia Moderada', description: 'Funcional con potencial de mejora' };
   175→  return { score, label: 'Contraste Alto', description: 'Combinacion atrevida y audaz' };
   176→}

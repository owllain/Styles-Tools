    46→function useFavorites() {
    47→  const [favs, setFavs] = useState<string[]>(() => {
    48→    if (typeof window === 'undefined') return [];
    49→    try { return JSON.parse(localStorage.getItem(FAVS_KEY) || '[]'); } catch { return []; }
    50→  });
    51→  const toggle = useCallback((id: string) => {
    52→    setFavs(prev => {
    53→      const next = prev.includes(id) ? prev.filter(f => f !== id) : [...prev, id];
    54→      localStorage.setItem(FAVS_KEY, JSON.stringify(next));
    55→      return next;
    56→    });
    57→  }, []);
    58→  const isFav = useCallback((id: string) => favs.includes(id), [favs]);
    59→  return { favs, toggle, isFav };
    60→}

    77→function useWornTracker() {
    78→  const [worn, setWorn] = useState<Record<string, WornEntry>>(() => {
    79→    if (typeof window === 'undefined') return {};
    80→    try { return JSON.parse(localStorage.getItem(WORN_KEY) || '{}'); } catch { return {}; }
    81→  });
    82→  const markWorn = useCallback((id: string) => {
    83→    setWorn(prev => {
    84→      const today = new Date().toISOString().split('T')[0];
    85→      const entry = prev[id] || { count: 0, lastWorn: '', dates: [] };
    86→      const updated = { ...entry, count: entry.count + 1, lastWorn: today, dates: [...new Set([...entry.dates, today])] };
    87→      const next = { ...prev, [id]: updated };
    88→      localStorage.setItem(WORN_KEY, JSON.stringify(next));
    89→      return next;
    90→    });
    91→  }, []);
    92→  return { worn, markWorn };

   118→function getHarmonyScore(colors: string[]): { score: number; label: string; description: string } {
   119→  if (colors.length < 2) return { score: 50, label: 'Neutro', description: 'Datos insuficientes' };
   120→  
   121→  const hexToHSL = (hex: string) => {
   122→    let r = parseInt(hex.slice(1, 3), 16) / 255;
   123→    let g = parseInt(hex.slice(3, 5), 16) / 255;
   124→    let b = parseInt(hex.slice(5, 7), 16) / 255;
   125→    const max = Math.max(r, g, b), min = Math.min(r, g, b);
   126→    let h = 0, s = 0, l = (max + min) / 2;
   127→    if (max !== min) {
   128→      const d = max - min;
   129→      s = l > 0.5 ? d / (2 - max - min) : d / (max + min);
   130→      if (max === r) h = ((g - b) / d + (g < b ? 6 : 0)) / 6;
   131→      else if (max === g) h = ((b - r) / d + 2) / 6;
   132→      else h = ((r - g) / d + 4) / 6;
   133→    }
   134→    return { h: h * 360, s: s * 100, l: l * 100 };
   135→  };
   136→
   137→  const hueDiff = (a: number, b: number) => Math.abs(((a - b + 540) % 360) - 180);
   138→
   139→  const hslColors = colors.map(hexToHSL);
   140→  const neutrals = hslColors.filter(c => c.s < 15);
   141→  const chromatics = hslColors.filter(c => c.s >= 15);
   142→
   143→  let score = 60; // base
   144→
   145→  // Bonus for neutrals (they go with everything)
   146→  score += Math.min(neutrals.length * 5, 10);
   147→
   148→  // Chromatic harmony
   149→  if (chromatics.length >= 2) {
   150→    const maxHueDiff = Math.max(...chromatics.flatMap((c, i) => chromatics.slice(i + 1).map(d => hueDiff(c.h, d.h))));
   151→    const avgSat = chromatics.reduce((sum, c) => sum + c.s, 0) / chromatics.length;
   152→    const avgLight = chromatics.reduce((sum, c) => sum + c.l, 0) / chromatics.length;
   153→
   154→    if (maxHueDiff < 30 || maxHueDiff > 330) score += 15; // Analogous
   155→    else if (Math.abs(maxHueDiff - 180) < 30) score += 12; // Complementary
   156→    else if (maxHueDiff > 90 && maxHueDiff < 270) score += 8; // Triadic area
   157→    else score += 5;
   158→
   159→    if (avgSat < 40) score += 5; // Muted palette bonus
   160→    if (avgLight > 20 && avgLight < 80) score += 5; // Good lightness range
   161→  }
   162→
   163→  // Monochrome bonus (very intentional)
   164→  const uniqueHues = new Set(hslColors.filter(c => c.s >= 15).map(c => Math.round(c.h / 15) * 15));
   165→  if (uniqueHues.size <= 2 && hslColors.length >= 3) score += 8;
   166→
   167→  score = Math.min(Math.max(score, 30), 98);
   168→
   169→  if (score >= 85) return { score, label: 'Armonia Excelente', description: 'Paleta cohesiva y sofisticada' };
   170→  if (score >= 70) return { score, label: 'Buena Armonia', description: 'Combinacion equilibrada y atractiva' };
   171→  if (score >= 55) return { score, label: 'Armonia Moderada', description: 'Funcional con potencial de mejora' };
   172→  return { score, label: 'Contraste Alto', description: 'Combinacion atrevida y audaz' };
   173→}
